    实战：企业知识库问答系统

    本章构建一个企业级知识库问答系统，支持多格式文档导入、增量更新、权限控制和高效检索。

    学习目标

    • 支持多格式文档（PDF、Word、网页）导入
    • 实现增量更新和文档版本管理
    • 优化检索策略（混合检索 + 重排序）
    • 添加引用来源展示

    1. 系统架构

    企业知识库系统的核心模块：

    • 文档管理：上传、解析、切分、向量化
    • 检索引擎：混合检索 + 重排序
    • 问答生成：基于检索结果生成回答
    • 引用展示：回答附带来源文档

    2. 多格式文档加载

    from langchain_community.document_loaders import (
    PyPDFLoader, Docx2txtLoader, WebBaseLoader, CSVLoader
)

def load_document(file_path):
    if file_path.endswith(".pdf"):
        return PyPDFLoader(file_path).load()
    elif file_path.endswith(".docx"):
        return Docx2txtLoader(file_path).load()
    elif file_path.endswith(".csv"):
        return CSVLoader(file_path).load()
    elif file_path.startswith("http"):
        return WebBaseLoader(file_path).load()
    else:
        from langchain_community.document_loaders import TextLoader
        return TextLoader(file_path).load()

    3. 文档切分与元数据

    from langchain.text_splitter import RecursiveCharacterTextSplitter
import hashlib

splitter = RecursiveCharacterTextSplitter(
    chunk_size=800,
    chunk_overlap=100,
    separators=["\n\n", "\n", "。", "！", "？", " ", ""]
)

def process_document(file_path, source_name):
    docs = load_document(file_path)
    chunks = splitter.split_documents(docs)
    doc_id = hashlib.md5(file_path.encode()).hexdigest()[:8]
    for chunk in chunks:
        chunk.metadata["doc_id"] = doc_id
        chunk.metadata["source"] = source_name
    return chunks, doc_id

    4. 向量库与增量更新

    from langchain_community.vectorstores import Chroma
from langchain_openai import OpenAIEmbeddings

vectorstore = Chroma(
    persist_directory="./kb_db",
    embedding_function=OpenAIEmbeddings()
)

def add_documents(chunks):
    vectorstore.add_documents(chunks)

def delete_document(doc_id):
    # 根据 metadata 中的 doc_id 删除
    vectorstore._collection.delete(
        where={"doc_id": doc_id}
    )

    5. 混合检索 + 重排序

    from langchain.retrievers import EnsembleRetriever, ContextualCompressionRetriever
from langchain_community.retrievers import BM25Retriever
from langchain.retrievers.document_compressors import LLMChainExtractor

bm25 = BM25Retriever.from_documents(all_chunks)
bm25.k = 10
vector_retriever = vectorstore.as_retriever(search_kwargs={"k": 10})

ensemble = EnsembleRetriever(
    retrievers=[bm25, vector_retriever],
    weights=[0.4, 0.6]
)

# 用 LLM 压缩，只保留相关片段
compressor = LLMChainExtractor.from_llm(llm)
retriever = ContextualCompressionRetriever(
    base_compressor=compressor,
    base_retriever=ensemble
)

    6. 问答链 + 引用来源

    from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser
from operator import itemgetter

prompt = ChatPromptTemplate.from_template("""
基于以下上下文回答用户问题。
如果上下文中没有答案，请回答"根据知识库无法回答该问题"。
回答末尾请列出引用来源。

上下文：
{context}

问题：{question}
""")

def format_with_sources(docs):
    context = ""
    sources = []
    for i, doc in enumerate(docs, 1):
        context += f"[{i}] {doc.page_content}\n"
        sources.append(f"[{i}] {doc.metadata.get('source', '未知')}")
    return context, sources

chain = (
    {
        "context": itemgetter("question") | retriever,
        "question": itemgetter("question"),
    }
    | prompt | llm | StrOutputParser()
)

def ask(question):
    docs = retriever.invoke(question)
    context, sources = format_with_sources(docs)
    answer = chain.invoke({"question": question, "context": context})
    return answer + "\n\n引用来源：\n" + "\n".join(sources)

    7. 权限控制（简化版）

    企业知识库常需权限控制，可通过 metadata 过滤实现：

    # 添加文档时设置权限
chunk.metadata["department"] = "研发部"

# 检索时按部门过滤
retriever = vectorstore.as_retriever(
    search_kwargs={
        "k": 5,
        "filter": {"department": "研发部"}
    }
)
